cmake_minimum_required(VERSION 3.20)
project(GlabDesktop VERSION 0.1.0 LANGUAGES CXX)
find_package(Tela CONFIG REQUIRED)
find_package(nlohmann_json 3.11 CONFIG REQUIRED)
add_library(glab_desktop STATIC
  src/desktop_ui.cpp src/screen_document.cpp src/screen_actions.cpp)
add_library(GLAB::Desktop ALIAS glab_desktop)
target_include_directories(glab_desktop PUBLIC include)
target_compile_features(glab_desktop PUBLIC cxx_std_20)
target_link_libraries(glab_desktop PUBLIC Tela::Core nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(glab_desktop PRIVATE /utf-8 /W4 /permissive-)
endif()
if(WIN32)
  if(NOT TARGET Tela::Pictor)
    message(FATAL_ERROR "The installed Tela package must include Tela::Pictor")
  endif()
  target_sources(glab_desktop PRIVATE src/resident_host.cpp src/native_form.cpp)
  target_compile_definitions(glab_desktop PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(glab_desktop PUBLIC Tela::Pictor PRIVATE user32 gdi32 shell32)
endif()
